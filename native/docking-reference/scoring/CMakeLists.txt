cmake_minimum_required(VERSION 3.20)
project(mole_docking_reference LANGUAGES CXX)
enable_testing()

add_library(mole_docking_reference STATIC src/scoring.cpp src/scoring_field.cpp src/search_objective.cpp)
target_include_directories(mole_docking_reference PUBLIC include)
target_compile_features(mole_docking_reference PUBLIC cxx_std_20)
if(MSVC)
  target_compile_options(mole_docking_reference PRIVATE /fp:strict /W4 /WX)
else()
  target_compile_options(mole_docking_reference PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off -fno-fast-math)
endif()

add_executable(mole_docking_reference_tests tests/scoring_test.cpp)
target_link_libraries(mole_docking_reference_tests PRIVATE mole_docking_reference)
target_compile_definitions(mole_docking_reference_tests PRIVATE MOLE_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../../../tests/fixtures")
add_test(NAME mole_docking_reference_tests COMMAND mole_docking_reference_tests)
add_executable(mole_docking_reference_field_tests tests/scoring_field_test.cpp)
target_link_libraries(mole_docking_reference_field_tests PRIVATE mole_docking_reference)
if(WIN32)
  target_link_libraries(mole_docking_reference_field_tests PRIVATE psapi)
endif()
add_test(NAME mole_docking_reference_field_tests COMMAND mole_docking_reference_field_tests)
if(MSVC)
  target_compile_options(mole_docking_reference_tests PRIVATE /fp:strict /W4 /WX)
else()
  target_compile_options(mole_docking_reference_tests PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off -fno-fast-math)
endif()
if(MSVC)
  target_compile_options(mole_docking_reference_field_tests PRIVATE /fp:strict /W4 /WX)
else()
  target_compile_options(mole_docking_reference_field_tests PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off -fno-fast-math)
endif()
add_executable(mole_docking_reference_gradient_tests tests/search_objective_test.cpp)
target_link_libraries(mole_docking_reference_gradient_tests PRIVATE mole_docking_reference)
add_test(NAME mole_docking_reference_gradient_tests COMMAND mole_docking_reference_gradient_tests)
if(MSVC)
  target_compile_options(mole_docking_reference_gradient_tests PRIVATE /fp:strict /W4 /WX)
else()
  target_compile_options(mole_docking_reference_gradient_tests PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off -fno-fast-math)
endif()
