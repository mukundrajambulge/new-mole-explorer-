cmake_minimum_required(VERSION 3.20)
project(mole_d3_fullpose_runner LANGUAGES CXX)
add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../../../native/docking-reference/scoring ${CMAKE_BINARY_DIR}/scoring-core EXCLUDE_FROM_ALL)
add_executable(mole_d3_fullpose_compare full_pose_compare.cpp)
target_link_libraries(mole_d3_fullpose_compare PRIVATE mole_docking_reference)
target_compile_features(mole_d3_fullpose_compare PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(mole_d3_fullpose_compare PRIVATE /fp:strict /W4 /WX)
else()
  target_compile_options(mole_d3_fullpose_compare PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off -fno-fast-math)
endif()
